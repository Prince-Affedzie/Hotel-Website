  <script>
    // NAV active state
    const navLinks = document.querySelectorAll('.nav-link');
    function onScroll(){
      const scrollPos = window.scrollY + 120;
      navLinks.forEach(a => {
        const el = document.querySelector(a.getAttribute('href'));
        if (!el) return;
        if (el.offsetTop <= scrollPos && el.offsetTop + el.offsetHeight > scrollPos) {
          a.classList.add('active');
        } else {
          a.classList.remove('active');
        }
      });
    }
    window.addEventListener('scroll', onScroll);
    onScroll();

    // Simple testimonials carousel
    const slides = document.getElementById('slides');
    const slideCount = slides.children.length;
    let index = 0;
    function showSlide(i){
      index = (i + slideCount) % slideCount;
      slides.style.transform = 'translateX(' + (-index * 100) + '%)';
    }
    document.getElementById('prevSlide').addEventListener('click', ()=> showSlide(index - 1));
    document.getElementById('nextSlide').addEventListener('click', ()=> showSlide(index + 1));
    let autoSlide = setInterval(()=> showSlide(index + 1), 6000);
    // Pause auto on hover
    slides.addEventListener('mouseenter', ()=> clearInterval(autoSlide));
    slides.addEventListener('mouseleave', ()=> autoSlide = setInterval(()=> showSlide(index + 1), 6000));

    // Booking form handler (client-side)
    document.getElementById('bookingForm').addEventListener('submit', function(e){
      e.preventDefault();
      const data = new FormData(this);
      // In production, send to server via fetch -> e.g. fetch('/api/book', {method:'POST', body: data})
      document.getElementById('bookingMessage').textContent = 'Thanks! We are checking availability. A reservations agent will contact you shortly.';
      this.reset();
    });

    // Prefill booking helper
    function prefillBooking(room){
      const form = document.getElementById('bookingForm');
      form.roomType.value = room;
      window.location.href = '#contact';
      // briefly highlight
      form.scrollIntoView({behavior:'smooth', block:'center'});
    }

    // Contact form handler
    document.getElementById('contactForm').addEventListener('submit', function(e){
      e.preventDefault();
      document.getElementById('contactMessage').textContent = 'Message sent! We will reply within 24 hours.';
      this.reset();
    });

    // Quick booking
    document.getElementById('quickForm').addEventListener('submit', function(e){
      e.preventDefault();
      document.getElementById('quickMessage').textContent = 'Booking request received. Our team will contact you to confirm.';
      this.reset();
    });

    // Small accessibility: allow Enter to submit on quick forms in some mobile contexts
    document.querySelectorAll('input, textarea').forEach(i=>{
      i.addEventListener('invalid', () => i.classList.add('invalid'));
    });

    // Replace "logo.png" gracefully: if not found, fallback to text
    document.querySelectorAll('.logo img').forEach(img=>{
      img.onerror = function(){ this.style.display='none'; };
    });
